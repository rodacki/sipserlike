  % ----------------------------------------------------
  % Espaços antes e depois do capítulo (oficiais!)
  % ----------------------------------------------------

  \setlength{\afterchapskip}{30pt}

  % ----------------------------------------------------

  % ----------------------------------------------------

  \renewcommand{\chapternamenum}{}

  % ----------------------------------------------------
  % Número do capítulo
  % ----------------------------------------------------
  \renewcommand{\printchapternum}{%
    {\Huge \CopperplateBold\addfontfeatures{LetterSpace=3.0}\thechapter}%
    \par\vspace{10pt}%
  }

  % ----------------------------------------------------
  % Título do capítulo
  % ----------------------------------------------------
  \renewcommand{\printchaptertitle}[1]{%
    {\Huge \CopperplateBold\addfontfeatures{LetterSpace=2.0}\MakeUppercase{##1}}%
  }

  % ----------------------------------------------------
  % Alinhamento à esquerda
  % ----------------------------------------------------
  \renewcommand{\chaptitlefont}{\Huge\normalfont\bfseries}
  \renewcommand{\chapnumfont}{\Huge\normalfont\bfseries}
  \renewcommand{\afterchapternum}{\par\vspace{10pt}}

}

% ---------- Fonte do número igual à do número da página ----------
\newcommand{\sipsernumfont}{%
  \begingroup
  \normalfont\sffamily\bfseries\fontsize{7pt}{7pt}\selectfont
}

\newcommand{\endsipsernumfont}{%
  \endgroup
}

% ---------- Pontinho central ----------
\newcommand{\sipserdot}{%
  \raisebox{0ex}{\scalebox{1.0}{\textperiodcentered}}%
}

\newcommand{\sipserdots}{%
  \leavevmode
  \leaders\hbox{\sipserdot\kern-2.2mm}\hfill\kern0pt
}

% ---------- Cabeçalho do teorema ----------
\newcommand{\sipserthmhead}[2]{%
  {\Copperplate\MakeUppercase{#1}}%
  \hspace{4mm}%
  {\sipsernumfont #2\endsipsernumfont}%
  \hspace{4mm}%
  \sipserdots
  \par
}

% ---------- THEOREMSTYLE ----------
\newtheoremstyle{sipserstyle}
  {10pt}   % space above
  {10pt}   % space below
  {}       % body font
  {}       % indent
  {}       % headfont
  {}       % punctuation
  {\newline}
  {\sipserthmhead{#1}{#2}}

\theoremstyle{sipserstyle}

\newtheorem{definition}{Definition}[chapter]
\newtheorem{theorem}{Theorem}[chapter]
